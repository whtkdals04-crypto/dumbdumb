  const UNITS = [

  const SANGMIN_COLS = ['정리', '쎄다', '혼합형', '심화별', 'Half', '단원별'];
  const SOI_COLS = ['유형별', '유형별+', '쎄다', '혼합형', '단원별', '심화별', 'HALF', '파트별 다지기'];
  const SOI_MARKS = ['', '—', '⊤', '下', '止', '正'];
  const SANGMIN_MARKS = ['', '|', '||', '|||', '||||', '正'];

  function getWeekInfo(d) {
    const date = new Date(d);
    date.setHours(0, 0, 0, 0);
    const day = date.getDay();
    const diffToMon = date.getDate() - day + (day === 0 ? -6 : 1);
    
    const mon = new Date(date.setDate(diffToMon));
    const sun = new Date(mon);
    sun.setDate(mon.getDate() + 6);

    const target = new Date(mon);
    target.setDate(target.getDate() + 3 - (target.getDay() + 6) % 7);
    const week1 = new Date(target.getFullYear(), 0, 4);
    const weekNum = 1 + Math.round(((target - week1) / 86400000 - 3 + (week1.getDay() + 6) % 7) / 7);
    const weekKey = `${target.getFullYear()}-W${String(weekNum).padStart(2, '0')}`;
    const rangeStr = `${mon.getMonth() + 1}월 ${mon.getDate()}일 ~ ${sun.getMonth() + 1}월 ${sun.getDate()}일`;

    return { weekKey, rangeStr, monDate: mon, sunDate: sun };
  }

  window.currentTab = 'sangmin';
  window.currentDate = new Date();
  window.currentWeekInfo = getWeekInfo(window.currentDate);
  window.currentColor = '#ffb7b2';
  window.archiveFilter = 'sangmin';
  window.calendarDate = new Date();

  function initProgressTables() {
    const sangminBody = document.getElementById('progressSangminBody');
    let sangminHtml = '';
    UNITS.forEach((unit, uIdx) => {
      sangminHtml += `<tr><td class="unit-name">${unit}</td>`;
      SANGMIN_COLS.forEach((col, cIdx) => {
        sangminHtml += `<td class="progress-cell" data-user="sangmin" data-unit="${uIdx}" data-col="${cIdx}" onclick="toggleProgressCell(this)"></td>`;
      });
      sangminHtml += `</tr>`;
    });
    sangminBody.innerHTML = sangminHtml;

    const soiBody = document.getElementById('progressSoiBody');
    let soiHtml = '';
    UNITS.forEach((unit, uIdx) => {
      soiHtml += `<tr><td class="unit-name">${unit}</td>`;
      SOI_COLS.forEach((col, cIdx) => {
        soiHtml += `<td class="progress-cell" data-user="soi" data-unit="${uIdx}" data-col="${cIdx}" onclick="toggleProgressCell(this)"></td>`;
      });
      soiHtml += `</tr>`;
    });
    soiBody.innerHTML = soiHtml;

    loadProgressData('sangmin');
    loadProgressData('soi');
  }

  function toggleProgressCell(td) {
    const user = td.dataset.user;
    const unit = td.dataset.unit;
    const col = td.dataset.col;
    let count = parseInt(td.dataset.count || '0', 10);

    if (user === 'sangmin' && parseInt(col, 10) === 0) {
      count = count === 1 ? 0 : 1;
    } else {
      count = (count + 1) % 6;
    }

    td.dataset.count = count;
    renderProgressCellText(td);
    saveProgressData(user);
  }

  function renderProgressCellText(td) {
    const user = td.dataset.user;
    const col = parseInt(td.dataset.col, 10);
    const count = parseInt(td.dataset.count || '0', 10);

    if (count === 0) {
      td.innerText = '';
      td.classList.remove('marked-sangmin', 'marked-soi');
      return;
    }

    if (user === 'sangmin') {
      td.classList.add('marked-sangmin');
      if (col === 0) {
        td.innerText = '✓';
      } else {
        td.innerText = SANGMIN_MARKS[count];
      }
    } else {
      td.classList.add('marked-soi');
      td.innerText = SOI_MARKS[count];
    }
  }

  function saveProgressData(user) {
    const data = {};
    document.querySelectorAll(`.progress-cell[data-user="${user}"]`).forEach(cell => {
      const key = `${cell.dataset.unit}_${cell.dataset.col}`;
      data[key] = cell.dataset.count || '0';
    });
    localStorage.setItem(`progress_${user}`, JSON.stringify(data));
  }

  function loadProgressData(user) {
    const raw = localStorage.getItem(`progress_${user}`);
    if (!raw) return;
    try {
      const data = JSON.parse(raw);
      Object.keys(data).forEach(key => {
        const [uIdx, cIdx] = key.split('_');
        const cell = document.querySelector(`.progress-cell[data-user="${user}"][data-unit="${uIdx}"][data-col="${cIdx}"]`);
        if (cell) {
          cell.dataset.count = data[key];
          renderProgressCellText(cell);
        }
      });
    } catch(e) { console.error(e); }
  }

  function resetSaveButtonState() {
    const saveBtn = document.getElementById('saveBtn');
    if (saveBtn) {
      saveBtn.classList.remove('saved');
      saveBtn.innerHTML = '💾 저장하기 (기출 보관함으로)';
    }
  }

  function setSavedButtonState() {
    const saveBtn = document.getElementById('saveBtn');
    if (saveBtn) {
      saveBtn.classList.add('saved');
      saveBtn.innerHTML = '✅ 저장완료';
    }
  }

  function onDataChanged() {
    resetSaveButtonState();
    savePlannerData(false);
  }

  function loadPlannerData() {
    if (window.currentTab === 'archive') {
      loadArchiveList();
      return;
    }

    resetSaveButtonState();
    const weekKey = window.currentWeekInfo.weekKey;
    document.getElementById('currentWeekLabel').innerText = `${window.currentWeekInfo.rangeStr} (${weekKey})`;
    
    const docId = `${window.currentTab}_${weekKey}`;
    const savedData = localStorage.getItem(`planner_${docId}`);
    
    if (savedData) {
      const data = JSON.parse(savedData);
      renderPlannerData(data);
      if (data.isArchived) {
        setSavedButtonState();
      }
    } else {
      clearPlannerUI();
    }
  }

  function savePlannerData(isExplicitSave = false) {
    if (['archive', 'progress-sangmin', 'progress-soi', 'calendar'].includes(window.currentTab)) return;

    const weekKey = window.currentWeekInfo.weekKey;
    const docId = `${window.currentTab}_${weekKey}`;

    const weeklyMemos = [];
    document.querySelectorAll('.weekly-memo').forEach(input => {
      weeklyMemos.push(input.value);
    });

    const todos = [];
    let totalTodoCount = 0;
    let completedTodoCount = 0;

    document.querySelectorAll('.todo-cell').forEach((cell) => {
      const items = [];
      cell.querySelectorAll('.todo-item').forEach(item => {
        const textVal = item.querySelector('input[type="text"]').value.trim();
        const isChecked = item.querySelector('input[type="checkbox"]').checked;
        items.push({
          checked: isChecked,
          text: textVal
        });
        if (textVal.length > 0) {
          totalTodoCount++;
          if (isChecked) completedTodoCount++;
        }
      });
      todos.push(items);
    });

    const completionRate = totalTodoCount > 0 
      ? Math.round((completedTodoCount / totalTodoCount) * 100) 
      : 0;

    const slotData = {};
    let totalPaintedSlots = 0;
    const daySlotCounts = [0, 0, 0, 0, 0, 0, 0];

    document.querySelectorAll('.slot-cell').forEach(slot => {
      if (slot.style.backgroundColor && slot.style.backgroundColor !== 'transparent' && slot.style.backgroundColor !== 'rgb(255, 255, 255)') {
        slotData[slot.dataset.slotId] = slot.style.backgroundColor;
        totalPaintedSlots++;
        
        const dayIdx = parseInt(slot.dataset.slotId.split('-')[0], 10);
        if (!isNaN(dayIdx)) daySlotCounts[dayIdx]++;
      }
    });

    const existingDoc = localStorage.getItem(`planner_${docId}`);
    const existing = existingDoc ? JSON.parse(existingDoc) : {};

    const payload = {
      id: docId,
      owner: window.currentTab,
      ownerName: window.currentTab === 'sangmin' ? '상민' : '소이',
      weekKey: weekKey,
      dateRange: window.currentWeekInfo.rangeStr,
      weeklyMemos: weeklyMemos,
      todos: todos,
      slots: slotData,
      totalSlots: totalPaintedSlots,
      totalMinutes: totalPaintedSlots * 10, // 10분 단위 계산
      daySlotCounts: daySlotCounts,
      totalTodoCount: totalTodoCount,
      completedTodoCount: completedTodoCount,
      completionRate: completionRate,
      updatedAt: new Date().toISOString(),
      isArchived: isExplicitSave ? true : (existing.isArchived || false)
    };

    localStorage.setItem(`planner_${docId}`, JSON.stringify(payload));

    if (isExplicitSave) {
      setSavedButtonState();
    }
  }

  function loadArchiveList() {
    const archiveContainer = document.getElementById('archiveList');
    archiveContainer.innerHTML = `<div style="text-align:center; padding: 40px; color: #64748b;">보관된 플래너 데이터를 불러오는 중...</div>`;

    let docsData = [];
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i);
      if (key && key.startsWith('planner_')) {
        try {
          const item = JSON.parse(localStorage.getItem(key));
          if (item && item.isArchived) {
            docsData.push(item);
          }
        } catch (e) {
          console.error(e);
        }
      }
    }

    docsData.sort((a, b) => (b.weekKey || '').localeCompare(a.weekKey || ''));

    if (window.archiveFilter !== 'all') {
      docsData = docsData.filter(item => item.owner === window.archiveFilter);
    }

    if (docsData.length === 0) {
      archiveContainer.innerHTML = `<div style="text-align:center; padding: 50px; color: #94a3b8; font-weight: 600;">저장된 플래너 기록이 없습니다.<br>플래너에서 '저장하기' 버튼을 눌러 기록을 남겨보세요!</div>`;
      return;
    }

    let html = '';
    docsData.forEach(item => {
      const hours = Math.floor((item.totalMinutes || 0) / 60);
      const mins = (item.totalMinutes || 0) % 60;
      const rate = item.completionRate !== undefined ? item.completionRate : 0;
      const ownerBadge = item.owner === 'sangmin' 
        ? '<span class="owner-tag sangmin">상민</span>' 
        : '<span class="owner-tag soi">소이</span>';

      html += `
        <div class="archive-card" onclick="openArchiveDetail('${item.id}')">
          <div class="archive-header">
            <div class="archive-title">${ownerBadge} ${item.dateRange}</div>
            <div class="archive-sub">${item.weekKey}</div>
          </div>
          <div class="archive-stats">
            <div class="stat-box">
              <div class="stat-label">총 공부 시간 (칸당 10분)</div>
              <div class="stat-value">${hours}시간 ${mins}분</div>
            </div>
            <div class="stat-box">
              <div class="stat-label">할 일 달성률</div>
              <div class="stat-value" style="color: #22c55e;">${rate}%</div>
            </div>
          </div>
          <div class="archive-footer">
            <span>클릭하여 상세 할 일 및 공부 시간 분석 보기 ➔</span>
          </div>
        </div>
      `;
    });

    archiveContainer.innerHTML = html;
    window.archiveDocsCache = docsData;
  }

  function openArchiveDetail(docId) {
    const item = (window.archiveDocsCache || []).find(d => d.id === docId);
    if (!item) return;

    const days = ['월', '화', '수', '목', '금', '토', '일'];
    const hours = Math.floor((item.totalMinutes || 0) / 60);
    const mins = (item.totalMinutes || 0) % 60;

    document.getElementById('modalTitle').innerText = `[${item.ownerName}] ${item.dateRange} 공부 기록`;
    document.getElementById('modalTotalTime').innerText = `${hours}시간 ${mins}분 (${item.totalSlots || 0}칸)`;

    let dayTimeHtml = '';
    if (item.daySlotCounts) {
      item.daySlotCounts.forEach((cnt, idx) => {
        const dMins = cnt * 10;
        const dH = Math.floor(dMins / 60);
        const dM = dMins % 60;
        dayTimeHtml += `<div class="day-time-pill"><b>${days[idx]}:</b> ${dH > 0 ? dH + '시간 ' : ''}${dM}분</div>`;
      });
    }
    document.getElementById('modalDayTimes').innerHTML = dayTimeHtml;

    let memoHtml = '';
    if (item.weeklyMemos) {
      item.weeklyMemos.forEach((m, idx) => {
        if (m.trim()) memoHtml += `<li><b>${idx + 1}.</b> ${m}</li>`;
      });
    }
    document.getElementById('modalMemos').innerHTML = memoHtml || '<li style="color:#94a3b8;">작성된 목표가 없습니다.</li>';

    let todoListHtml = '';
    if (item.todos) {
      item.todos.forEach((dayTodos, dIdx) => {
        let itemsHtml = '';
        dayTodos.forEach(t => {
          if (t.text.trim()) {
            itemsHtml += `
              <div class="modal-todo-item ${t.checked ? 'completed' : ''}">
                <span class="checkbox-icon">${t.checked ? '✓' : '○'}</span>
                <span class="todo-text">${t.text}</span>
              </div>
            `;
          }
        });

        if (itemsHtml) {
          todoListHtml += `
            <div class="modal-day-group">
              <div class="modal-day-title">${days[dIdx]}요일</div>
              ${itemsHtml}
            </div>
          `;
        }
      });
    }
    document.getElementById('modalTodoList').innerHTML = todoListHtml || '<div style="color:#94a3b8; padding:10px;">기록된 할 일이 없습니다.</div>';

    document.getElementById('detailModal').style.display = 'flex';
  }

  function closeArchiveDetail() {
    document.getElementById('detailModal').style.display = 'none';
  }

  function filterArchive(filter) {
    window.archiveFilter = filter;
    document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
    document.getElementById(`filter-${filter}`).classList.add('active');
    loadArchiveList();

  function renderPlannerData(data) {
    clearPlannerUI();
    if (data.weeklyMemos) {
      document.querySelectorAll('.weekly-memo').forEach((input, idx) => {
        if (data.weeklyMemos[idx]) input.value = data.weeklyMemos[idx];
      });
    }
    if (data.todos) {
      document.querySelectorAll('.todo-cell').forEach((cell, cellIdx) => {
        const items = cell.querySelectorAll('.todo-item');
        if (data.todos[cellIdx]) {
          data.todos[cellIdx].forEach((itemData, itemIdx) => {
            if (items[itemIdx]) {
              items[itemIdx].querySelector('input[type="checkbox"]').checked = itemData.checked;
              items[itemIdx].querySelector('input[type="text"]').value = itemData.text;
            }
          });
        }
      });
    }
    if (data.slots) {
      Object.keys(data.slots).forEach(id => {
        const slot = document.querySelector(`[data-slot-id="${id}"]`);
        if (slot) slot.style.backgroundColor = data.slots[id];
      });
    }
  }

  function clearPlannerUI() {
    document.querySelectorAll('.weekly-memo').forEach(i => i.value = '');
    document.querySelectorAll('.slot-cell').forEach(s => s.style.backgroundColor = '');
    document.querySelectorAll('.todo-item input[type="text"]').forEach(i => i.value = '');
    document.querySelectorAll('.todo-item input[type="checkbox"]').forEach(c => c.checked = false);
  }

  function renderCalendar() {
    const year = window.calendarDate.getFullYear();
    const month = window.calendarDate.getMonth();

    document.getElementById('calendarYearMonth').innerText = `${year}년 ${month + 1}월`;

    const firstDay = new Date(year, month, 1);
    const lastDay = new Date(year, month + 1, 0);

    const startDayOfWeek = firstDay.getDay(); 
    const totalDays = lastDay.getDate();

    const prevMonthLastDay = new Date(year, month, 0).getDate();

    const grid = document.getElementById('calendarDaysGrid');
    grid.innerHTML = '';

    const events = getExamEvents();

    for (let i = startDayOfWeek - 1; i >= 0; i--) {
      const d = prevMonthLastDay - i;
      const cell = document.createElement('div');
      cell.className = 'calendar-cell other-month';
      cell.innerHTML = `<div class="cell-date">${d}</div>`;
      grid.appendChild(cell);
    }

    const today = new Date();
    for (let d = 1; d <= totalDays; d++) {
      const cell = document.createElement('div');
      const isToday = today.getFullYear() === year && today.getMonth() === month && today.getDate() === d;
      
      const currentCellDate = new Date(year, month, d);
      const dayOfWeek = currentCellDate.getDay();

      cell.className = `calendar-cell ${isToday ? 'today' : ''}`;
      
      let dateClass = '';
      if (dayOfWeek === 0) dateClass = 'sun';
      else if (dayOfWeek === 6) dateClass = 'sat';

      const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
      cell.onclick = () => openExamModal(dateStr);

      let cellHtml = `<div class="cell-date ${dateClass}">${d}</div>`;

      const dayEvents = events.filter(e => e.date === dateStr);
      dayEvents.forEach(e => {
        let badgeClass = '';
        if (e.type === 'quiz') badgeClass = 'quiz';
        else if (e.type === 'assignment') badgeClass = 'assignment';
        
        cellHtml += `<div class="exam-badge ${badgeClass}" title="${e.title}">${e.title}</div>`;
      });

      cell.innerHTML = cellHtml;
      grid.appendChild(cell);
    }

    const remainingCells = (7 - ((startDayOfWeek + totalDays) % 7)) % 7;
    for (let d = 1; d <= remainingCells; d++) {
      const cell = document.createElement('div');
      cell.className = 'calendar-cell other-month';
      cell.innerHTML = `<div class="cell-date">${d}</div>`;
      grid.appendChild(cell);
    }

    renderExamCards();
  }

  function moveCalendarMonth(delta) {
    window.calendarDate.setMonth(window.calendarDate.getMonth() + delta);
    renderCalendar();
  }

  function goTodayCalendar() {
    window.calendarDate = new Date();
    renderCalendar();
  }

  function getExamEvents() {
    const raw = localStorage.getItem('calendar_exams');
    if (!raw) return [];
    try { return JSON.parse(raw); } catch(e) { return []; }
  }

  function openExamModal(selectedDate) {
    if (selectedDate) {
      document.getElementById('examDateInput').value = selectedDate;
    } else {
      const today = new Date();
      const dateStr = `${today.getFullYear()}-${String(today.getMonth() + 1).padStart(2, '0')}-${String(today.getDate()).padStart(2, '0')}`;
      document.getElementById('examDateInput').value = dateStr;
    }
    document.getElementById('examModal').style.display = 'flex';
  }

  function closeExamModal() {
    document.getElementById('examModal').style.display = 'none';
    document.getElementById('examTitleInput').value = '';
    document.getElementById('examMemoInput').value = '';
  }

  function saveExamEvent() {
    const title = document.getElementById('examTitleInput').value.trim();
    const date = document.getElementById('examDateInput').value;
    const type = document.getElementById('examTypeInput').value;
    const memo = document.getElementById('examMemoInput').value.trim();

    if (!title || !date) {
      alert('과목명/시험명과 날짜를 입력해주세요.');
      return;
    }

    const events = getExamEvents();
    events.push({
      id: 'exam_' + Date.now(),
      type, title, date, memo
    });

    localStorage.setItem('calendar_exams', JSON.stringify(events));
    closeExamModal();
    renderCalendar();
  }

  function deleteExamEvent(id) {
    let events = getExamEvents();
    events = events.filter(e => e.id !== id);
    localStorage.setItem('calendar_exams', JSON.stringify(events));
    renderCalendar();
  }

  function renderExamCards() {
    const events = getExamEvents();
    const container = document.getElementById('examCardGrid');

    if (events.length === 0) {
      container.innerHTML = `<div style="grid-column: 1 / -1; color:#94a3b8; font-size:12px; font-weight:600;">등록된 시험 및 학교 일정이 없습니다.</div>`;
      return;
    }

    events.sort((a,b) => a.date.localeCompare(b.date));

    const today = new Date();
    today.setHours(0,0,0,0);

    let html = '';
    events.forEach(e => {
      const eDate = new Date(e.date);
      eDate.setHours(0,0,0,0);

      const diffTime = eDate - today;
      const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

      let ddayText = '';
      let isPassed = false;
      if (diffDays === 0) {
        ddayText = 'D-DAY';
      } else if (diffDays > 0) {
        ddayText = `D-${diffDays}`;
      } else {
        ddayText = `D+${Math.abs(diffDays)}`;
        isPassed = true;
      }

      let typeName = '시험';
      if (e.type === 'quiz') typeName = '퀴즈';
      if (e.type === 'assignment') typeName = '과제';

      html += `
        <div class="exam-card">
          <div class="exam-dday ${isPassed ? 'passed' : ''}">${ddayText}</div>
          <div class="exam-card-title">[${typeName}] ${e.title}</div>
          <div class="exam-card-date">📅 ${e.date}</div>
          ${e.memo ? `<div style="font-size:11px; color:#475569; margin-top:6px; background:#fff; padding:6px; border-radius:4px; border:1px solid #cbd5e1;">${e.memo}</div>` : ''}
          <button onclick="deleteExamEvent('${e.id}')" style="margin-top:8px; background:none; border:none; color:#ef4444; font-size:10px; font-weight:700; cursor:pointer; padding:0;">삭제</button>
        </div>
      `;
    });

    container.innerHTML = html;
  }

  function switchTab(tab) {
    window.currentTab = tab;

    document.querySelectorAll('.tab-button').forEach(btn => btn.classList.remove('active'));
    document.getElementById(`tab-${tab}`).classList.add('active');

    const plannerView = document.getElementById('plannerView');
    const progressSangminView = document.getElementById('progressSangminView');
    const progressSoiView = document.getElementById('progressSoiView');
    const archiveView = document.getElementById('archiveView');
    const calendarView = document.getElementById('calendarView');
    const ownerBadge = document.getElementById('plannerOwnerBadge');

    plannerView.style.display = 'none';
    progressSangminView.style.display = 'none';
    progressSoiView.style.display = 'none';
    archiveView.style.display = 'none';
    calendarView.style.display = 'none';

    if (tab === 'archive') {
      archiveView.style.display = 'block';
      loadArchiveList();
    } else if (tab === 'progress-sangmin') {
      progressSangminView.style.display = 'block';
    } else if (tab === 'progress-soi') {
      progressSoiView.style.display = 'block';
    } else if (tab === 'calendar') {
      calendarView.style.display = 'block';
      renderCalendar();
    } else {
      plannerView.style.display = 'block';
      if (tab === 'sangmin') {
        ownerBadge.innerText = 'PLANNER (상민)';
        ownerBadge.style.color = 'var(--primary-sangmin)';
      } else {
        ownerBadge.innerText = 'PLANNER (소이)';
        ownerBadge.style.color = 'var(--primary-soi)';
      }
      loadPlannerData();
    }
  }

  let isTouching = false;

  window.addEventListener('touchstart', (e) => {
    if(e.target.classList.contains('slot-cell')) { isTouching = true; paint(e.target); }
  }, { passive: true });

  window.addEventListener('touchmove', (e) => {
    if (!isTouching) return;
    const touch = e.touches[0];
    const target = document.elementFromPoint(touch.clientX, touch.clientY);
    if (target && target.classList.contains('slot-cell')) { paint(target); }
  }, { passive: true });

  window.addEventListener('touchend', () => { if(isTouching) { isTouching = false; onDataChanged(); } });
  
  document.querySelectorAll('.slot-cell').forEach(cell => {
    cell.addEventListener('mousedown', () => { isTouching = true; paint(cell); });
    cell.addEventListener('mouseenter', () => { if (isTouching) paint(cell); });
  });
  window.addEventListener('mouseup', () => { if(isTouching) { isTouching = false; onDataChanged(); } });
